cmake_minimum_required(VERSION 3.22)
project(bunkercrew_audio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(BUNKERCREW_EXTENSION "Build Godot extension" ON)
set(TESSEL third_party/tessel/src)
add_library(tessel_engine STATIC ${TESSEL}/audio/DrumEngine.cpp ${TESSEL}/audio/EffectsRack.cpp ${TESSEL}/audio/TonalVoice.cpp ${TESSEL}/generator/BeatGenerator.cpp ${TESSEL}/generator/TonalComposer.cpp)
target_include_directories(tessel_engine PUBLIC ${TESSEL})
add_library(bunkercrew_music STATIC src/live_music.cpp)
target_include_directories(bunkercrew_music PUBLIC src)
target_link_libraries(bunkercrew_music PUBLIC tessel_engine)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(tessel_engine PRIVATE -O3 -ffp-contract=off)
 target_compile_options(bunkercrew_music PRIVATE -O3 -ffp-contract=off)
endif()
add_executable(bunkercrew_audio_tests tests/music_tests.cpp)
find_package(Threads REQUIRED)
target_link_libraries(bunkercrew_audio_tests PRIVATE bunkercrew_music Threads::Threads)
if(WIN32)
 target_link_options(bunkercrew_audio_tests PRIVATE -static -static-libgcc -static-libstdc++ -Wl,--stack,16777216)
endif()
enable_testing()
add_test(NAME live_music COMMAND bunkercrew_audio_tests)
if(BUNKERCREW_EXTENSION)
 set(GODOTCPP_CUSTOM_API_FILE "${CMAKE_CURRENT_SOURCE_DIR}/api/extension_api.json" CACHE FILEPATH "" FORCE)
 set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE FILEPATH "" FORCE)
 set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
 if(NOT GODOT_CPP_PATH)
  message(FATAL_ERROR "Set GODOT_CPP_PATH to pinned godot-cpp checkout")
 endif()
 add_subdirectory("${GODOT_CPP_PATH}" godot-cpp)
 add_library(bunkercrew_audio SHARED src/tessel_stream.cpp src/register_types.cpp)
 target_link_libraries(bunkercrew_audio PRIVATE bunkercrew_music godot::cpp)
 target_compile_definitions(bunkercrew_audio PRIVATE TESSEL_BUILD_GODOT=1)
 if(WIN32)
  set_target_properties(bunkercrew_audio PROPERTIES PREFIX "" SUFFIX ".windows.x86_64.dll")
  target_link_options(bunkercrew_audio PRIVATE -static -static-libgcc -static-libstdc++)
 else()
  set_target_properties(bunkercrew_audio PROPERTIES PREFIX "lib" SUFFIX ".linux.x86_64.so")
 endif()
endif()
